cmake_minimum_required(VERSION 3.20)
project(SmartAutocomplete VERSION 1.0 LANGUAGES CXX)
include(CTest)
option(SMART_BUILD_SERVER "Build the Crow web application" ON)
option(SMART_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  if(SMART_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(smart_core src/core/text.cpp src/core/dictionary.cpp src/core/algorithms.cpp)
target_include_directories(smart_core PUBLIC include)
if(SMART_BUILD_SERVER)
  include(FetchContent)
  set(CROW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CROW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(CROW_ENABLE_SSL OFF CACHE BOOL "" FORCE)
  set(CROW_ENABLE_COMPRESSION OFF CACHE BOOL "" FORCE)
  set(CROW_INSTALL OFF CACHE BOOL "" FORCE)
  set(CROW_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/Crow-v1.3.4.tar.gz" CACHE FILEPATH "Cached pinned Crow archive")
  if(EXISTS "${CROW_ARCHIVE}")
    set(CROW_URL "${CROW_ARCHIVE}")
  else()
    set(CROW_URL "https://codeload.github.com/CrowCpp/Crow/tar.gz/refs/tags/v1.3.4")
  endif()
  FetchContent_Declare(Crow URL "${CROW_URL}" URL_HASH SHA256=6be759b3a648b41f0adc3a8408a53fd0243775217b4e32e8101d69e4aadf48cc DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(Crow)
  add_library(smart_api src/web/api.cpp)
  target_include_directories(smart_api PUBLIC src/web)
  target_link_libraries(smart_api PUBLIC smart_core Crow::Crow)
  add_executable(smartautocomplete src/main.cpp)
  target_link_libraries(smartautocomplete PRIVATE smart_api)
  target_compile_definitions(smartautocomplete PRIVATE CROW_DISABLE_STATIC_DIR SMART_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  if(BUILD_TESTING)
    add_executable(api_tests tests/api_tests.cpp)
    target_link_libraries(api_tests PRIVATE smart_api)
    add_test(NAME api COMMAND api_tests)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE smart_core)
  target_compile_definitions(core_tests PRIVATE SMART_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME core COMMAND core_tests)
endif()
